45// which mode a world is currently running in. A processor only runs if its phase
46// allows the mode the world is in.

48{

50 PlayMode = 1, // world is running gameplay
51};

56{

341inline void
342World::SetComponent(Entity entity, TYPE const& value)
343{
344#if NEBULA_DEBUG
345 n_assert2(
346 !this->pipeline.IsRunningAsync(),
347 "Settings components in arbitrary entities while executing an async processor is currently not supported!"
348 );
349#endif
350
351#if NEBULA_DEBUG
352 n_assert2(
353 sizeof(TYPE) == MemDb::AttributeRegistry::TypeSize(Game::GetComponentId<TYPE>()),
354 "SetComponent: Provided value's type is not the correct size for the given ComponentId."
355 );
356#endif
357 EntityMapping mapping = this->GetEntityMapping(entity);
358 TYPE* ptr = (TYPE*)this->GetInstanceBuffer(mapping.table, mapping.instance.partition, GetComponentId<TYPE>());
359 *(ptr + mapping.instance.index) = value;
360}

417//------------------------------------------------------------------------------
420template <typename TYPE>
421inline TYPE*
422World::AddComponent(Entity entity)
423{
424 /*
425 We could possibly support this by either having thread local allocators and
426 staged queues and gathering after the processors have finished (must clear
427 allocators by issuing a separate job for this at the end of the frame), or by
428 just introducing a simple mutex.
429 */
430#if NEBULA_DEBUG
431 n_assert2(
432 !this->pipeline.IsRunningAsync(), "Adding component to entities while in an async processor is currently not supported!"
433 );
434#endif
435 Game::ComponentId id = Game::GetComponentId<TYPE>();
436#if NEBULA_DEBUG
437 n_assert(MemDb::AttributeRegistry::TypeSize(id) == sizeof(TYPE));
438 //n_assert(!this->HasComponent<TYPE>(entity));
439#endif
440 TYPE* data = this->componentStageAllocator.Alloc<TYPE>();
441 const void* defaultValue = MemDb::AttributeRegistry::DefaultValue(id);
442 Memory::Copy(defaultValue, data, sizeof(TYPE));
443
444 AddStagedComponentCommand cmd = {
445 .entity = entity,
446 .componentId = id,
447 .dataSize = sizeof(TYPE),
448 .data = data,
449 };
450 this->addStagedQueue.Append(cmd);
451
452 MemDb::Attribute* attr = MemDb::AttributeRegistry::GetAttribute(id);
453 ComponentInterface* cInterface;
454 cInterface = static_cast<ComponentInterface*>(attr);
455
456 if (this->componentInitializationEnabled && cInterface->Init != nullptr)
457 {
458 // run initialization function if it exists.
459 cInterface->Init(this, entity, data);
460 }
461
462 return data;
463}
