415//------------------------------------------------------------------------------
418__forceinline scalar
419determinant(const mat4& m)
420{
421 __m128 Va,Vb,Vc;
422 __m128 r1,r2,r3,tt,tt2;
423 __m128 sum,Det;
424
425 __m128 _L1 = m.r[0].vec;
426 __m128 _L2 = m.r[1].vec;
427 __m128 _L3 = m.r[2].vec;
428 __m128 _L4 = m.r[3].vec;
429 // Calculating the minterms for the first line.
430
431 // _mm_ror_ps is just a macro using _mm_shuffle_ps().
432 tt = _L4; tt2 = mm_ror_ps(_L3,1);
433 Vc = _mm_mul_ps(tt2, mm_ror_ps(tt,0)); // V3' dot V4
434 Va = _mm_mul_ps(tt2, mm_ror_ps(tt,2)); // V3' dot V4"
435 Vb = _mm_mul_ps(tt2, mm_ror_ps(tt,3)); // V3' dot V4^
436
437 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V3" dot V4^ - V3^ dot V4"
438 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V3^ dot V4' - V3' dot V4^
439 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V3' dot V4" - V3" dot V4'
440
441 tt = _L2;
442 Va = mm_ror_ps(tt,1); sum = _mm_mul_ps(Va,r1);
443 Vb = mm_ror_ps(tt,2); sum = _mm_add_ps(sum,_mm_mul_ps(Vb,r2));
444 Vc = mm_ror_ps(tt,3); sum = _mm_add_ps(sum,_mm_mul_ps(Vc,r3));
445
446 // Calculating the determinant.
447 Det = _mm_mul_ps(sum,_L1);
448 Det = _mm_add_ps(Det,_mm_movehl_ps(Det,Det));
449
450 // Calculating the minterms of the second line (using previous results).
451 tt = mm_ror_ps(_L1,1); sum = _mm_mul_ps(tt,r1);
452 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
453 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
454
455 // Testing the determinant.
456 Det = _mm_sub_ss(Det,_mm_shuffle_ps(Det,Det,1));
457 return vec4(Det).x;
458}

460//------------------------------------------------------------------------------
463__forceinline mat4
464transpose(const mat4& m)
465{
466 __m128 tmp3, tmp2, tmp1, tmp0;
467 tmp0 = _mm_unpacklo_ps(m.r[0].vec, m.r[1].vec);
468 tmp2 = _mm_unpacklo_ps(m.r[2].vec, m.r[3].vec);
469 tmp1 = _mm_unpackhi_ps(m.r[0].vec, m.r[1].vec);
470 tmp3 = _mm_unpackhi_ps(m.r[2].vec, m.r[3].vec);
471 mat4 ret;
472 ret.r[0] = _mm_movelh_ps(tmp0, tmp2);
473 ret.r[1] = _mm_movehl_ps(tmp2, tmp0);
474 ret.r[2] = _mm_movelh_ps(tmp1, tmp3);
475 ret.r[3] = _mm_movehl_ps(tmp3, tmp1);
476 return ret;
477}

479//------------------------------------------------------------------------------
482__forceinline mat4
483inverse(const mat4& m)
484{
485 __m128 Va,Vb,Vc;
486 __m128 r1,r2,r3,tt,tt2;
487 __m128 sum,Det,RDet;
488 __m128 trns0,trns1,trns2,trns3;
489
490 const __m128i pnpn = _mm_setr_epi32(0x00000000, static_cast<int>(0x80000000), 0x00000000, static_cast<int>(0x80000000));
491 const __m128i npnp = _mm_setr_epi32(static_cast<int>(0x80000000), 0x00000000, static_cast<int>(0x80000000), 0x00000000);
492 const __m128 zeroone = _mm_setr_ps(1.0f, 0.0f, 0.0f, 1.0f);
493
494 __m128 _L1 = m.r[0].vec;
495 __m128 _L2 = m.r[1].vec;
496 __m128 _L3 = m.r[2].vec;
497 __m128 _L4 = m.r[3].vec;
498 // Calculating the minterms for the first line.
499
500 // _mm_ror_ps is just a macro using _mm_shuffle_ps().
501 tt = _L4; tt2 = mm_ror_ps(_L3,1);
502 Vc = _mm_mul_ps(tt2, mm_ror_ps(tt,0)); // V3'dot V4
503 Va = _mm_mul_ps(tt2, mm_ror_ps(tt,2)); // V3'dot V4"
504 Vb = _mm_mul_ps(tt2, mm_ror_ps(tt,3)); // V3' dot V4^
505
506 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V3" dot V4^ - V3^ dot V4"
507 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V3^ dot V4' - V3' dot V4^
508 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V3' dot V4" - V3" dot V4'
509
510 tt = _L2;
511 Va = mm_ror_ps(tt,1); sum = _mm_mul_ps(Va,r1);
512 Vb = mm_ror_ps(tt,2); sum = _mm_add_ps(sum,_mm_mul_ps(Vb,r2));
513 Vc = mm_ror_ps(tt,3); sum = _mm_add_ps(sum,_mm_mul_ps(Vc,r3));
514
515 // Calculating the determinant.
516 Det = _mm_mul_ps(sum,_L1);
517 Det = _mm_add_ps(Det,_mm_movehl_ps(Det,Det));
518
519
520 __m128 mtL1 = _mm_xor_ps(sum, _mm_castsi128_ps(pnpn));
521
522 // Calculating the minterms of the second line (using previous results).
523 tt = mm_ror_ps(_L1,1); sum = _mm_mul_ps(tt,r1);
524 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
525 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
526 __m128 mtL2 = _mm_xor_ps(sum, _mm_castsi128_ps(npnp));
527
528 // Testing the determinant.
529 Det = _mm_sub_ss(Det,_mm_shuffle_ps(Det,Det,1));
530
531 // Calculating the minterms of the third line.
532 tt = mm_ror_ps(_L1,1);
533 Va = _mm_mul_ps(tt,Vb); // V1' dot V2"
534 Vb = _mm_mul_ps(tt,Vc); // V1' dot V2^
535 Vc = _mm_mul_ps(tt,_L2); // V1' dot V2
536
537 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V1" dot V2^ - V1^ dot V2"
538 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V1^ dot V2' - V1' dot V2^
539 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V1' dot V2" - V1" dot V2'
540
541 tt = mm_ror_ps(_L4,1); sum = _mm_mul_ps(tt,r1);
542 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
543 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
544 __m128 mtL3 = _mm_xor_ps(sum, _mm_castsi128_ps(pnpn));
545
546 // Dividing is FASTER than rcp_nr! (Because rcp_nr causes many register-memory RWs).
547 RDet = _mm_div_ss(zeroone, Det); // TODO: just 1.0f?
548 RDet = _mm_shuffle_ps(RDet,RDet,0x00);
549
550 // Devide the first 12 minterms with the determinant.
551 mtL1 = _mm_mul_ps(mtL1, RDet);
552 mtL2 = _mm_mul_ps(mtL2, RDet);
553 mtL3 = _mm_mul_ps(mtL3, RDet);
554
555 // Calculate the minterms of the forth line and devide by the determinant.
556 tt = mm_ror_ps(_L3,1); sum = _mm_mul_ps(tt,r1);
557 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
558 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
559 __m128 mtL4 = _mm_xor_ps(sum, _mm_castsi128_ps(npnp));
560 mtL4 = _mm_mul_ps(mtL4, RDet);
561
562 // Now we just have to transpose the minterms matrix.
563 trns0 = _mm_unpacklo_ps(mtL1,mtL2);
564 trns1 = _mm_unpacklo_ps(mtL3,mtL4);
565 trns2 = _mm_unpackhi_ps(mtL1,mtL2);
566 trns3 = _mm_unpackhi_ps(mtL3,mtL4);
567 _L1 = _mm_movelh_ps(trns0,trns1);
568 _L2 = _mm_movehl_ps(trns1,trns0);
569 _L3 = _mm_movelh_ps(trns2,trns3);
570 _L4 = _mm_movehl_ps(trns3,trns2);
571
572 return mat4(vec4(_L1), vec4(_L2), vec4(_L3), vec4(_L4));
573}

578__forceinline mat4
579lookatlh(const point& eye, const point& at, const vector& up)
580{
581#if NEBULA_DEBUG
582 n_assert(length(up) > 0);
583#endif
584 // hmm the XM lookat functions are kinda pointless, because they
585 // return a VIEW matrix, which is already inverse (so one would
586 // need to reverse again!)
587 const vector zaxis = normalize(at - eye);
588 vector normUp = normalize(up);
589 if (Math::abs(dot(zaxis, normUp)) > 0.9999999f)
590 {
591 // need to choose a different up vector because up and lookat point
592 // into same or opposite direction
593 // just rotate y->x, x->z and z->y
594 normUp = permute(normUp, normUp, 1, 2, 0);
595 }
596 const vector xaxis = normalize(cross(normUp, zaxis));
597 const vector yaxis = normalize(cross(zaxis, xaxis));
598 mat4 ret;
599 ret.row0 = xaxis;
600 ret.row0.w = -dot(xaxis, eye);
601 ret.row1 = yaxis;
602 ret.row1.w = -dot(yaxis, eye);
603 ret.row2 = zaxis;
604 ret.row2.w = -dot(zaxis, eye);
605 ret.row3 = vec4(0, 0, 0, 1);
606 return transpose(ret);
607}

644#ifdef N_USE_AVX
645// dual linear combination using AVX instructions on YMM regs
646static inline __m256 twolincomb_AVX_8(__m256 A01, const mat4 &B)
647{
648 __m256 result;
649 result = _mm256_mul_ps(_mm256_shuffle_ps(A01, A01, 0x00), _mm256_broadcast_ps(&B.r[0].vec));
650 result = _mm256_add_ps(result, _mm256_mul_ps(_mm256_shuffle_ps(A01, A01, 0x55), _mm256_broadcast_ps(&B.r[1].vec)));
651 result = _mm256_add_ps(result, _mm256_mul_ps(_mm256_shuffle_ps(A01, A01, 0xaa), _mm256_broadcast_ps(&B.r[2].vec)));
652 result = _mm256_add_ps(result, _mm256_mul_ps(_mm256_shuffle_ps(A01, A01, 0xff), _mm256_broadcast_ps(&B.r[3].vec)));
653 return result;
654}
