561__forceinline scalar
562determinant(const mat4& m)
563{
564 __m128 Va,Vb,Vc;
565 __m128 r1,r2,r3,tt,tt2;
566 __m128 sum,Det;
567
568 __m128 _L1 = m.r[0].vec;
569 __m128 _L2 = m.r[1].vec;
570 __m128 _L3 = m.r[2].vec;
571 __m128 _L4 = m.r[3].vec;
572 // Calculating the minterms for the first line.
573
574 // _mm_ror_ps is just a macro using _mm_shuffle_ps().
575 tt = _L4; tt2 = mm_ror_ps(_L3,1);
576 Vc = _mm_mul_ps(tt2, mm_ror_ps(tt,0)); // V3' dot V4
577 Va = _mm_mul_ps(tt2, mm_ror_ps(tt,2)); // V3' dot V4"
578 Vb = _mm_mul_ps(tt2, mm_ror_ps(tt,3)); // V3' dot V4^
579
580 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V3" dot V4^ - V3^ dot V4"
581 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V3^ dot V4' - V3' dot V4^
582 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V3' dot V4" - V3" dot V4'
583
584 tt = _L2;
585 Va = mm_ror_ps(tt,1); sum = _mm_mul_ps(Va,r1);
586 Vb = mm_ror_ps(tt,2); sum = _mm_add_ps(sum,_mm_mul_ps(Vb,r2));
587 Vc = mm_ror_ps(tt,3); sum = _mm_add_ps(sum,_mm_mul_ps(Vc,r3));
588
589 // Calculating the determinant.
590 Det = _mm_mul_ps(sum,_L1);
591 Det = _mm_add_ps(Det,_mm_movehl_ps(Det,Det));
592
593 // Calculating the minterms of the second line (using previous results).
594 tt = mm_ror_ps(_L1,1); sum = _mm_mul_ps(tt,r1);
595 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
596 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
597
598 // Testing the determinant.
599 Det = _mm_sub_ss(Det,_mm_shuffle_ps(Det,Det,1));
600 return vec4(Det).x;
601}

606__forceinline mat4
607transpose(const mat4& m)
608{
609 __m128 tmp3, tmp2, tmp1, tmp0;
610 tmp0 = _mm_unpacklo_ps(m.r[0].vec, m.r[1].vec);
611 tmp2 = _mm_unpacklo_ps(m.r[2].vec, m.r[3].vec);
612 tmp1 = _mm_unpackhi_ps(m.r[0].vec, m.r[1].vec);
613 tmp3 = _mm_unpackhi_ps(m.r[2].vec, m.r[3].vec);
614 mat4 ret;
615 ret.r[0] = _mm_movelh_ps(tmp0, tmp2);
616 ret.r[1] = _mm_movehl_ps(tmp2, tmp0);
617 ret.r[2] = _mm_movelh_ps(tmp1, tmp3);
618 ret.r[3] = _mm_movehl_ps(tmp3, tmp1);
619 return ret;
620}

625__forceinline mat4
626inverse(const mat4& m)
627{
628 __m128 Va,Vb,Vc;
629 __m128 r1,r2,r3,tt,tt2;
630 __m128 sum,Det,RDet;
631 __m128 trns0,trns1,trns2,trns3;
632
633 const __m128i pnpn = _mm_setr_epi32(0x00000000, static_cast<int>(0x80000000), 0x00000000, static_cast<int>(0x80000000));
634 const __m128i npnp = _mm_setr_epi32(static_cast<int>(0x80000000), 0x00000000, static_cast<int>(0x80000000), 0x00000000);
635 const __m128 zeroone = _mm_setr_ps(1.0f, 0.0f, 0.0f, 1.0f);
636
637 __m128 _L1 = m.r[0].vec;
638 __m128 _L2 = m.r[1].vec;
639 __m128 _L3 = m.r[2].vec;
640 __m128 _L4 = m.r[3].vec;
641 // Calculating the minterms for the first line.
642
643 // _mm_ror_ps is just a macro using _mm_shuffle_ps().
644 tt = _L4; tt2 = mm_ror_ps(_L3,1);
645 Vc = _mm_mul_ps(tt2, mm_ror_ps(tt,0)); // V3'dot V4
646 Va = _mm_mul_ps(tt2, mm_ror_ps(tt,2)); // V3'dot V4"
647 Vb = _mm_mul_ps(tt2, mm_ror_ps(tt,3)); // V3' dot V4^
648
649 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V3" dot V4^ - V3^ dot V4"
650 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V3^ dot V4' - V3' dot V4^
651 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V3' dot V4" - V3" dot V4'
652
653 tt = _L2;
654 Va = mm_ror_ps(tt,1); sum = _mm_mul_ps(Va,r1);
655 Vb = mm_ror_ps(tt,2); sum = _mm_add_ps(sum,_mm_mul_ps(Vb,r2));
656 Vc = mm_ror_ps(tt,3); sum = _mm_add_ps(sum,_mm_mul_ps(Vc,r3));
657
658 // Calculating the determinant.
659 Det = _mm_mul_ps(sum,_L1);
660 Det = _mm_add_ps(Det,_mm_movehl_ps(Det,Det));
661
662
663 __m128 mtL1 = _mm_xor_ps(sum, _mm_castsi128_ps(pnpn));
664
665 // Calculating the minterms of the second line (using previous results).
666 tt = mm_ror_ps(_L1,1); sum = _mm_mul_ps(tt,r1);
667 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
668 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
669 __m128 mtL2 = _mm_xor_ps(sum, _mm_castsi128_ps(npnp));
670
671 // Testing the determinant.
672 Det = _mm_sub_ss(Det,_mm_shuffle_ps(Det,Det,1));
673
674 // Calculating the minterms of the third line.
675 tt = mm_ror_ps(_L1,1);
676 Va = _mm_mul_ps(tt,Vb); // V1' dot V2"
677 Vb = _mm_mul_ps(tt,Vc); // V1' dot V2^
678 Vc = _mm_mul_ps(tt,_L2); // V1' dot V2
679
680 r1 = _mm_sub_ps(mm_ror_ps(Va,1), mm_ror_ps(Vc,2)); // V1" dot V2^ - V1^ dot V2"
681 r2 = _mm_sub_ps(mm_ror_ps(Vb,2), mm_ror_ps(Vb,0)); // V1^ dot V2' - V1' dot V2^
682 r3 = _mm_sub_ps(mm_ror_ps(Va,0), mm_ror_ps(Vc,1)); // V1' dot V2" - V1" dot V2'
683
684 tt = mm_ror_ps(_L4,1); sum = _mm_mul_ps(tt,r1);
685 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
686 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
687 __m128 mtL3 = _mm_xor_ps(sum, _mm_castsi128_ps(pnpn));
688
689 // Dividing is FASTER than rcp_nr! (Because rcp_nr causes many register-memory RWs).
690 RDet = _mm_div_ss(zeroone, Det); // TODO: just 1.0f?
691 RDet = _mm_shuffle_ps(RDet,RDet,0x00);
692
693 // Devide the first 12 minterms with the determinant.
694 mtL1 = _mm_mul_ps(mtL1, RDet);
695 mtL2 = _mm_mul_ps(mtL2, RDet);
696 mtL3 = _mm_mul_ps(mtL3, RDet);
697
698 // Calculate the minterms of the forth line and devide by the determinant.
699 tt = mm_ror_ps(_L3,1); sum = _mm_mul_ps(tt,r1);
700 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r2));
701 tt = mm_ror_ps(tt,1); sum = _mm_add_ps(sum,_mm_mul_ps(tt,r3));
702 __m128 mtL4 = _mm_xor_ps(sum, _mm_castsi128_ps(npnp));
703 mtL4 = _mm_mul_ps(mtL4, RDet);
704
705 // Now we just have to transpose the minterms matrix.
706 trns0 = _mm_unpacklo_ps(mtL1,mtL2);
707 trns1 = _mm_unpacklo_ps(mtL3,mtL4);
708 trns2 = _mm_unpackhi_ps(mtL1,mtL2);
709 trns3 = _mm_unpackhi_ps(mtL3,mtL4);
710 _L1 = _mm_movelh_ps(trns0,trns1);
711 _L2 = _mm_movehl_ps(trns1,trns0);
712 _L3 = _mm_movelh_ps(trns2,trns3);
713 _L4 = _mm_movehl_ps(trns3,trns2);
714
715 return mat4(vec4(_L1), vec4(_L2), vec4(_L3), vec4(_L4));
716}

722lookatlh(const point& eye, const point& at, const vector& up)
723{
724#if NEBULA_DEBUG
725 n_assert(length(up) > 0);
726#endif
727 // hmm the XM lookat functions are kinda pointless, because they
728 // return a VIEW matrix, which is already inverse (so one would
729 // need to reverse again!)
730 const vector zaxis = normalize(at - eye);
731 vector normUp = normalize(up);
732 if (Math::abs(dot(zaxis, normUp)) > 0.9999999f)
733 {
734 // need to choose a different up vector because up and lookat point
735 // into same or opposite direction
736 // just rotate y->x, x->z and z->y
737 normUp = permute(normUp, normUp, 1, 2, 0);
738 }
739 const vector xaxis = normalize(cross(normUp, zaxis));
740 const vector yaxis = normalize(cross(zaxis, xaxis));
741 mat4 ret;
742 ret.row0() = xaxis;
743 ret.row0().w = -dot(xaxis, eye);
744 ret.row1() = yaxis;
745 ret.row1().w = -dot(yaxis, eye);
746 ret.row2() = zaxis;
747 ret.row2().w = -dot(zaxis, eye);
748 ret.row3() = vec4(0, 0, 0, 1);
749 return transpose(ret);
750}

769 // into same or opposite direction
770 // just rotate y->x, x->z and z->y
771 normUp = permute(normUp, normUp, 1, 2, 0);
772 }
773 const vector xaxis = normalize(cross(normUp, zaxis));
774 const vector yaxis = normalize(cross(zaxis, xaxis));
775
776 mat4 ret;
777 ret.row0() = xaxis;
778 ret.row0().w = -dot(xaxis, eye);
779 ret.row1() = yaxis;
780 ret.row1().w = -dot(yaxis, eye);
781 ret.row2() = zaxis;
782 ret.row2().w = -dot(zaxis, eye);
783 ret.row3() = vec4(0, 0, 0, 1);
784 return transpose(ret);
785}

1158{

1160

1162 scalar cangle = Math::cos(angle);

1165 __m128 c = _mm_set_ps1(cangle);
1166 __m128 s = _mm_set_ps1(sangle);
1167
1168 __m128 nn1 = _mm_shuffle_ps(norm,norm,_MM_SHUFFLE(3,0,2,1));
1169 __m128 nn2 = _mm_shuffle_ps(norm,norm,_MM_SHUFFLE(3,1,0,2));
1170
1171 __m128 v = _mm_mul_ps(nn1,m1_c);
1172 v = _mm_mul_ps(nn2,v);
1173
1174 __m128 nn3 = _mm_mul_ps(norm, m1_c);
1175 nn3 = _mm_mul_ps(norm, nn3);
1176 nn3 = _mm_add_ps(nn3, c);
1177
1178 __m128 nn4 = _mm_mul_ps(norm,s);
1179 nn4 = _mm_add_ps(nn4, v);
1180 __m128 nn5 = _mm_mul_ps(s, norm);
1181 nn5 = _mm_sub_ps(v,nn5);
1182
1183 v = _mm_and_ps(nn3, _mask_xyz);
1184
1185 __m128 v1 = _mm_shuffle_ps(nn4,nn5,_MM_SHUFFLE(2,1,2,0));

1190
1191
