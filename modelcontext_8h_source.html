307inline void
308ModelContext::Dealloc(Graphics::ContextEntityId id)
309{
310 // clean up old stuff, but don't deallocate entity
311 Resources::ResourceId rid = modelContextAllocator.Get<Model_Id>(id.id);
312
313 if (rid != Resources::InvalidResourceId) // decrement model resource
314 Resources::DiscardResource(rid);
315 rid = Resources::InvalidResourceId;
316
317 modelContextAllocator.Get<Model_NodeInstanceRoots>(id.id).Clear();
318 modelContextAllocator.Get<Model_NodeLookup>(id.id).Clear();
319
320 // Deallocate material instances
321 const auto renderableRange = modelContextAllocator.Get<Model_NodeInstanceStates>(id.id).allocation;
322 for (size_t i = renderableRange.offset; i < renderableRange.offset + renderableRange.size; i++)
323 {
324 Materials::DestroyMaterialInstance(NodeInstances.renderable.nodeStates[(uint)i].materialInstance);
325 }
326 TransformInstanceAllocator.Dealloc(modelContextAllocator.Get<Model_NodeInstanceTransform>(id.id).allocation);
327 RenderInstanceAllocator.Dealloc(modelContextAllocator.Get<Model_NodeInstanceStates>(id.id).allocation);
328
329 modelContextAllocator.Dealloc(id.id);
330}
