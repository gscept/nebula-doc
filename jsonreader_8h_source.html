221template<typename T, size_t N>
222inline void
223JsonReader::Get(std::array<T, N>& ret, const char* attr)
224{
225 const pjson::value_variant* node = this->GetChild(attr);
226
227 n_assert(node->is_array());
228 n_assert(node->size() == N);
229
230 if constexpr (N > 0)
231 {
232 if (attr != nullptr)
233 {
234 this->SetToNode(attr);
235 }
236
237 bool validChild = this->SetToFirstChild();
238 n_assert(validChild);
239 for (size_t i = 0; i < N; i++)
240 {
241 this->Get<T>(ret[i]);
242 if (i + 1 < N)
243 {
244 validChild = this->SetToNextChild();
245 n_assert(validChild);
246 }
247 }
248 this->SetToParent();
249
250 if (attr != nullptr)
251 {
252 this->SetToParent();
253 }
254 }
255}

262JsonReader::Get(Util::BitField<N>& ret, const char* attr)
263{
264 Util::Array<int> arr;
265 this->Get<Util::Array<int>>(arr);

267 unsigned int count = arr.Size();
268 n_assert(count <= N);

271 ret.SetBit(arr[i]);
