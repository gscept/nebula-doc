145 , std::function<void(const Resources::ResourceId)> success
146 , std::function<void(const Resources::ResourceId)> failed
147 , bool immediate
148 , bool stream
149)
150{
151 // get resource loader by extension
152 Util::String ext = res.AsString().GetFileExtension();
153 IndexT i = this->extensionMap.FindIndex(ext);
154 n_warn_fmt(i != InvalidIndex, "No resource loader is associated with file extension '%s'", ext.AsCharPtr());
155 if (i == InvalidIndex)
